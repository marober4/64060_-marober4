fetch_modes <-function(Hitters){
uniq_hitters <- unique(Hitters)
freq_values <-tabulate(match(Hitters,uniq_hitters))
modes_years <- uniq_hitters[freq_values == max(freq_values)]
return(modes_years)
}

#plot histogram of home runs (hmrun)

ggplot(Hitters, aes(x = HmRun)) +
geom_histogram(binwidth = 5, fill = "orange", color = "blue") +
labs(title = " Home Runs across Hitters", x = "HmRun", y = "Count") +
theme_minimal()

#Histogram interpretation: more hitters (slightly less than 100 total) hit 5 home runs than any number #scatterplot of years to home runs. As the home runs count increases, the volume of players reaching the higher home run number decreases

ggplot(Hitters, aes(x = Years, y = HmRun)) +
geom_point(size = 1.5, color = "red") +
labs( title = "Scatter plot of Home Runs over Years", x = "Years of Playing", y = "Home Runs")

#Interpretation: years playing seems to increase the number of home runs for a player. In the early years, most of the home run dats is less than 10 homruns for years 0-6. After year 6, the number of home runs is distributed over a larger range with a higher mean and median than early years. For players with 20+ yars, the number of homeruns decreases again.
